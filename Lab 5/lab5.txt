// 1. What time is it?


#include "library.h"


void what_time_is_it()
{
        int time = get_clock_time();
        int date = get_calendar_date();
        int year = date / 10000;
        int month = (date / 100) % 100;
        int day = date % 100;
        int hour = time / 10000;
        int pm = hour >= 12;
        int minute = (time / 100) % 100;


        cout << "Year: " << year << endl; 
        cout << "Month: " << month << endl; 
        cout << "Day: " << day << endl; 
        cout << "Hour: " << hour << endl; 
        cout << "Minute: " << minute << endl; 
}


void main()
{
        what_time_is_it();
}

// 3. A Clock Face


#include "library.h"


void circle(double x, double y, double radius, const int steps)
{
        double pi = acos(-1.0);
        double circumference = 2 *pi *radius;
        double step_length = circumference/steps;
        double turn_angle = 360/steps;


        move_to(x + radius, y);
        set_heading_degrees(90);
        


        for (int i = 0; i < steps; i++) 
        {
                draw_distance(step_length);
                turn_right_by_degrees(turn_angle);
    }


        
}


void clock_marks(int x, int y)
{
        
        double radius = 200; 
        double steps = 360; 


        for (int i = 0; i < 12; i++) 
        {
                move_to(x,y);
        move_distance(radius - 15);
        draw_distance(15);
        move_distance(-(radius - 15));
        turn_right_by_degrees(30);
        }
}




void clock()
{

        clock_marks(260,320);
}

string month_name(int month)
{
        if (month == 1) return "January"; 
        if (month == 2) return "February";
        if (month == 3) return "March"; 
        if (month == 4) return "April"; 
        if (month == 5) return "May"; 
        if (month == 6) return "June"; 
        if (month == 7) return "July"; 
        if (month == 8) return "August";
        if (month == 9) return "September";
        if (month == 10) return "October"; 
        if (month == 11) return "November"; 
        if (month == 12) return "December";
}




string day_suffix(int day)
{
    if (day % 10 == 1 && day != 11) return "st";
    if (day % 10 == 2 && day != 12) return "nd";
    if (day % 10 == 3 && day != 13) return "rd";
    return "th";
}


void display_date_and_time(int text_size)
{
        int time = get_clock_time();
        int date = get_calendar_date();
        int year = date / 10000;
        int month = (date / 100) % 100;
        int day = date % 100;
        int hour = time / 10000;
        int minute = (time / 100) % 100;
        int pm = hour >= 12;
                if (hour > 12)
                        (hour = hour - 12);
                { 
                        if (hour >= 12) pm = true;
                }


        set_font_size(text_size);
        move_to(20,60);
        write_string(day); write_string(day_suffix(day)); write_string(" "); write_string(month_name(month)); write_string(" "); write_string(year);
        move_to(20,590);
        write_string(hour); write_string(":"); write_string(minute); 
        write_string(pm? " p.m.":" a.m.");
}
